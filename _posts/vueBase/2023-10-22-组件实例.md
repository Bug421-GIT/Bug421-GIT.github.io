- [$nextTick()](https://cn.vuejs.org/api/component-instance.html#nexttick) ：在下一次DOM更新结束后，执行其指定的回调

  1. Vue在更新DOM时是异步实现的 — 异步任务在同步任务完成之后才能执行，
  2. <u>Vue中改变响应式状态时，开启一个缓存异步任务的队列，直到下一个“tick”才一起执行</u> 
     - 确保每个组件无论发生多少状态改变，都仅执行一次更新（DOM 重新计算渲染）。
  3. 改变数据后，DOM需要重新更新，并且要基于更新后的新DOM进行某些操作
     - `$nextTick()`在状态改变后（DOM渲染后） 立即使用，以确保操作对象是存在、状态已更新

  - 全局 `Vue.nextTick(callFunction, this)`
  - 当前组件 `this.$nextTick(callFunction)`   — 回调函数中的 `this` 将自动绑定到当前的 `Vue` 实例上

  `$nextTick()` 会返回一个 `Promise` 对象

```js
// 一个回调函数作为参数
this.$refs.inputTitle.disabled = false // 打开输入框
this.$nextTick(function(){
    this.$refs.inputTitle.focus() // 同步执行时，关闭的输入框无法获取焦点
})

// await 返回的 Promise
async increment() {
      this.count++
      // DOM 还未更新
      console.log(document.getElementById('counter').textContent) // 0
      await nextTick()
      // DOM 此时已经更新
      console.log(document.getElementById('counter').textContent) // 1
}
```

4. 定义一个定时器 — 将操作放到异步队列（promise）中，等同步事件执行完毕再执行，原理一样（即使不给时间）

5. Vue2 中的源码判断逻辑：是否有promist - mutationObserve -setimmediate - setTimeout

   Vue3 中的源码 直接认为是promise，不是则返回一个promise

```ts
/* vue3.4 */
// resolvedPromise 也是一个promise
const resolvedPromise = /*#__PURE__*/ Promise.resolve() as Promise<any>
// currentFlushPromise 是一个promise	
let currentFlushPromise: Promise<void> | null = null
export function nextTick<T = void, R = void>(
  this: T, // 给 This 确定类型
  fn?: (this: T) => R
): Promise<Awaited<R>> {
  const p = currentFlushPromise || resolvedPromise
  return fn ? p.then(this ? fn.bind(this) : fn) : p // 将传入的回调放入promise中
}

// 调度异步任务
const queue: SchedulerJob[] = [] 
let isFlushing = false
export function queueJob(job: SchedulerJob) {
  // 重复数据消除搜索使用 Array.includes()的startIndex参数。默认情况下，搜索索引包括正在运行的当前作业，因此它不能再次递归触发自己。如果作业是watch（）回调，则搜索将从+1索引开始，以允许它递归地触发自己——用户有责任确保它不会以无限循环结束。
  // 去重
  if (job.id == null || !queue.includes(
      job, isFlushing && job.allowRecurse ? flushIndex + 1 : flushIndex
    )) {
      // 没有组件id 添加到数组末尾
    if (job.id == null) {
      queue.push(job)
    } else {// 按照jod.id 自增顺序排序
      queue.splice(findInsertionIndex(job.id), 0, job)
    }
    queueFlush()
  }
}

// 创建微任务
function queueFlush() {
  if (!isFlushing && !isFlushPending) { // 不挂起 && 不更新
    isFlushPending = true // 将任务挂起
    // 创建微任务 promise，将flushJobs放入执行
    currentFlushPromise = resolvedPromise.then(flushJobs)
  }
}

function flushJobs(seen?: CountMap) {
  isFlushPending = false // Pending 挂起 不
  isFlushing = true // Flushing 刷新
    
  queue.sort(comparator) // 队列重新排队
    // 1. 确保组件更新是由父组件到子组件——父组件在子组件之前创建，拥有更小更优先的编号
    // 2. 父组件的更新卸载了子组件，子组件的更新就可以被跳过了
    
  try { // 遍历队列，批量执行
    for (flushIndex = 0; flushIndex < queue.length; flushIndex++) {
      const job = queue[flushIndex]
      if (job && job.active !== false) {
        if (__DEV__ && check(job)) {
          continue
        }
        // console.log(`running:`, job.id)
        callWithErrorHandling(job, null, ErrorCodes.SCHEDULER) // 执行！
      }
    }
  } finally { // 执行完成后，重置状态
    flushIndex = 0
    queue.length = 0
	/*
        watch(xxx,()=>{},{
          flush:'post'
        })
         走 flushPostFlushCbs() 方法
  	*/
    flushPostFlushCbs(seen)

    isFlushing = false
    currentFlushPromise = null
    // some postFlushCb queued jobs!
    // keep flushing until it drains.
    if (queue.length || pendingPostFlushCbs.length) {
      flushJobs(seen)
    }
  }
}
```
